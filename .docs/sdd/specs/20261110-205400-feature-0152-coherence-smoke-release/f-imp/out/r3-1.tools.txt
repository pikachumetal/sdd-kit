const hourlyRates = { Norte: 20, Sur: 15 };
const weekendDays = ['sáb', 'dom'];
const weekendSurcharge = 1.5;

export function priceFor(room, day, hours) {
  if (!Object.hasOwn(hourlyRates, room)) return `sala desconocida: ${room}`;
  const surcharge = weekendDays.includes(day) ? weekendSurcharge : 1;
  const total = Math.round(hourlyRates[room] * Number(hours) * surcharge);
  return `${total} €`;
}
