import { execFileSync } from 'node:child_process';
import { readdirSync, readFileSync } from 'node:fs';
import { expect, test } from 'vitest';

// Cada título E2E actual está en «Flujos core» o en «Títulos heredados» de
// operations.md §Testing (sdd-tdd/references/writing-e2e.md, regla 1).

function titlesOf(suite, path = []) {
  const here = suite.title && !suite.file ? [...path, suite.title] : path;
  const own = (suite.specs ?? []).map((s) => [...here, s.title].join(' › '));
  const nested = (suite.suites ?? []).flatMap((s) => titlesOf(s, here));
  return [...own, ...nested];
}

function listedTitles() {
  const configs = readdirSync('.').filter((f) => /^playwright.*\.config\./.test(f));
  const titles = new Set();
  for (const config of configs) {
    const out = execFileSync(
      'npx',
      ['playwright', 'test', '--list', '--reporter=json', '-c', config],
      { encoding: 'utf8', shell: process.platform === 'win32', env: { ...process.env, PW_TEST_GREP: '' } },
    );
    const json = JSON.parse(out);
    if (json.errors?.length) throw new Error(JSON.stringify(json.errors));
    for (const s of json.suites) for (const t of titlesOf(s)) titles.add(t);
  }
  return titles;
}

function declaredTitles() {
  const ops = readFileSync('.docs/sdd/steering/operations.md', 'utf8').split('\n');
  const titles = new Set();
  let inInherited = false;
  for (const line of ops) {
    if (/\*\*Títulos heredados\*\*/.test(line)) { inInherited = true; continue; }
    if (/^- \*\*/.test(line) || /^## /.test(line)) inInherited = false;
    if (inInherited && /^\s+- /.test(line)) titles.add(line.replace(/^\s+- /, '').trim());
    if (/^\s+- .* · `e2e\//.test(line)) titles.add(line.split('·').pop().trim());
  }
  return titles;
}

test('cada título E2E está en Flujos core o en Títulos heredados', () => {
  const declared = declaredTitles();
  for (const title of listedTitles()) expect(declared, title).toContain(title);
});

test('un título heredado no desaparece sin quitarse de la lista', () => {
  const listed = listedTitles();
  const ops = readFileSync('.docs/sdd/steering/operations.md', 'utf8');
  const inherited = ops.split('**Títulos heredados**:')[1]?.split(/\n- \*\*/)[0] ?? '';
  for (const m of inherited.matchAll(/^\s+- (.+)$/gm)) expect(listed, m[1]).toContain(m[1].trim());
});
