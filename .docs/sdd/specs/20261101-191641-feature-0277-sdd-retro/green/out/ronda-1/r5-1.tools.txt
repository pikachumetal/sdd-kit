import { test } from 'node:test';
import assert from 'node:assert/strict';
import { readdirSync, readFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { join } from 'node:path';

const src = fileURLToPath(new URL('../src/', import.meta.url));
// Única salida permitida: la del CLI, que imprime el resultado de run().
const PERMITIDO = /console\.log\(run\(/;

test('src/ no deja console.log de depuración', () => {
  const culpables = [];
  for (const f of readdirSync(src, { recursive: true }).filter((f) => f.endsWith('.js'))) {
    readFileSync(join(src, f), 'utf8').split('\n').forEach((l, i) => {
      if (/\bconsole\.log\s*\(/.test(l) && !PERMITIDO.test(l)) culpables.push(`${f}:${i + 1}`);
    });
  }
  assert.deepEqual(culpables, []);
});
EOF
