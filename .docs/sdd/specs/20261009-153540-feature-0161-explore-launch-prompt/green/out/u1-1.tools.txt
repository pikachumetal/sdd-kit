const rooms = ['Norte', 'Sur'];
const bookings = [{ room: 'Norte', day: 'lun', slot: '10:00-12:00' }];

function freeRooms(slot) {
  return rooms.filter((room) => !bookings.some((b) => b.room === room && b.slot === slot));
}

export function run(cmd, params) {
  if (cmd === 'libres') return freeRooms(params[0]).join(', ');
  if (cmd === 'reservar' && params.includes('--cada-semana')) return 'reserva semanal creada';
  if (cmd === 'cancelar') return `cancelada ${params[0]} ${params[1]}`;
  return 'salas';
}

**Por qué esto reordena todo lo demás**

Una lista de espera es una cola de personas que compiten por la misma sala. Con un único usuario, esa cola no tiene sentido salvo que haya otra fuente de ocupación o que «espera» signifique otra cosa. Según la respuesta, cambian por completo:
- quién se pone en la cola,
- qué dispara el paso de espera a reserva,
- dónde viven los datos,
- si la lista de espera es antes o después de los avisos por correo del roadmap.
