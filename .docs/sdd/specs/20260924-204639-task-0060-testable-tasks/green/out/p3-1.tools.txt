import { test, beforeEach } from 'node:test';
import assert from 'node:assert';
import { request } from './helpers.js';

const names = (rooms) => rooms.map((r) => r.name);

async function roomId(name) {
  const rooms = await request('GET', '/rooms', { user: 'ana' });
  return rooms.find((r) => r.name === name).id;
}

beforeEach(async () => {
  for (const user of ['ana', 'luis']) {
    for (const name of ['Norte', 'Sur']) {
      await request('DELETE', `/rooms/${await roomId(name)}/favorite`, { user });
    }
  }
});

test('lista las salas sin favoritas con sus plazas y la estrella vacía', async () => {
  const rooms = await request('GET', '/rooms', { user: 'ana' });
  assert.deepStrictEqual(
    rooms.map(({ name, seats, favorite }) => ({ name, seats, favorite })),
    [
      { name: 'Norte', seats: 4, favorite: false },
      { name: 'Sur', seats: 12, favorite: false },
    ],
  );
});

test('marcar Sur la deja favorita y lo sigue siendo al volver a pedir la lista', async () => {
  await request('PUT', `/rooms/${await roomId('Sur')}/favorite`, { user: 'ana' });
  const rooms = await request('GET', '/rooms', { user: 'ana' });
  assert.strictEqual(rooms.find((r) => r.name === 'Sur').favorite, true);
});

});
