import { writeFileSync } from 'node:fs';
import { performance } from 'node:perf_hooks';
import { SLOT, loadApp, seed } from './harness.mjs';

const CALLS_PER_BATCH = 200;
const THRESHOLD_MS = 50;
const REPETITIONS = 12;

async function loadSeededApp() {
  const { run, bookings } = await loadApp();
  seed(bookings);
  const active = bookings.filter((b) => b.slot === SLOT && b.status === 'active').length;
  if (active !== 1000 || run('libres', [SLOT]) !== 'Sur') {
    console.error(`siembra incorrecta: ${active} activas en la franja, libres=${run('libres', [SLOT])}`);
    process.exit(1);
  }
  return run;
}

function timeCall(run) {
  const start = performance.now();
  run('libres', [SLOT]);
  return performance.now() - start;
}

function timeBatch(run) {
  return Array.from({ length: CALLS_PER_BATCH }, () => timeCall(run));
}

const median = (values) => [...values].sort((a, b) => a - b)[Math.floor(values.length / 2)];
const fmt = (ms) => ms.toFixed(4);
const evidenceUrl = (name) => new URL(name, import.meta.url);

async function measureO1() {
  const run = await loadSeededApp();
  const cold = timeCall(run);
  const calls = timeBatch(run);
  const max = Math.max(cold, ...calls);
  const lines = [`fría=${fmt(cold)}`, ...calls.map(fmt), `O1 medido: max=${fmt(max)} ms (umbral ${THRESHOLD_MS} ms)`];
  writeFileSync(evidenceUrl('evidencia-o1.txt'), `${lines.join('\n')}\n`);
  console.log(lines.at(-1));
}

async function measureEndurance() {
  if (!globalThis.gc) {
    console.error('el modo aguante necesita node --expose-gc');
    process.exit(1);
  }
  const run = await loadSeededApp();
  const heapMb = () => {
    globalThis.gc();
    return process.memoryUsage().heapUsed / 1024 / 1024;
  };
  const rows = [];
  let max = 0;
  const firstHeap = heapMb();
  for (let rep = 1; rep <= REPETITIONS; rep++) {
    const calls = timeBatch(run);
    max = Math.max(max, ...calls);
    rows.push(`rep=${rep} mediana=${fmt(median(calls))} max=${fmt(Math.max(...calls))} heap=${heapMb().toFixed(2)}`);

  const lastHeap = heapMb();
  const summary = `Aguante: ${REPETITIONS} repeticiones; max=${fmt(max)} ms; heap ${firstHeap.toFixed(2)} → ${lastHeap.toFixed(2)} MB`;
  writeFileSync(evidenceUrl('evidencia-aguante.txt'), `${[...rows, summary].join('\n')}\n`);
  console.log(summary);
}
