const groupings = {
  sala: {
    groups: () => [...rooms].sort(),
    groupOf: (b) => b.room,
    describe: (b) => `  ${b.day} ${b.slot} — ${b.reason} (${b.owner})`,
  },
  motivo: {
    groups: () => reasons,
    groupOf: (b) => b.reason,
    describe: (b) => `  ${b.room} ${b.day} ${b.slot} (${b.owner})`,
  },
};

function byRoomAndDay(a, b) {
  return rooms.indexOf(a.room) - rooms.indexOf(b.room) || week.indexOf(a.day) - week.indexOf(b.day);
}

function informe(params) {
  const grouping = groupings[params[params.indexOf('--por') + 1]];
  if (!grouping) return 'uso: informe --por sala|motivo';
  const cancelled = bookings.filter((b) => b.status === 'cancelled').sort(byRoomAndDay);
  const lines = [];
  for (const group of grouping.groups()) {
    const members = cancelled.filter((b) => grouping.groupOf(b) === group);
    if (!members.length) continue;
    lines.push(`${group}: ${members.length} canceladas`, ...members.map(grouping.describe));
  }
  lines.push(`Total: ${cancelled.length} canceladas`);
  return lines.join('\n');
}
EOS
