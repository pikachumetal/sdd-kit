import { expect, test, Browser, Page } from '@playwright/test';

// Estrés de la 0027 (edición a la vez), flujo core «editar celdas de una hoja y que se guarden».
// La rejilla tiene 4 columnas × 5 filas = 20 celdas, así que 5 sesiones editan 4 celdas cada una (su fila).
const SHEET = 'cierre';
const SESSIONS = 5;
const COLUMNS = ['A', 'B', 'C', 'D'];

async function openSession(browser: Browser): Promise<Page> {
  const page = await (await browser.newContext()).newPage();
  await page.goto(`/dev/login?user=agente@hojas.test&next=/hojas/${SHEET}`);
  await expect(page.getByRole('grid', { name: 'Hoja' })).toBeVisible();
  return page;
}

async function typeInCell(page: Page, cell: string, text: string) {
  await page.getByTestId(`cell-${cell}`).click();
  await page.keyboard.press('Enter');
  const editor = page.getByRole('textbox', { name: 'Editor de celda' });
  await expect(editor).toBeFocused();
  await editor.pressSequentially(text);
  await page.keyboard.press('Enter');
  await expect(page.getByTestId(`cell-${cell}`)).toHaveText(text);
}

test('cinco sesiones editan a la vez la hoja cierre y las celdas se guardan', async ({ browser, request }) => {
  const pages = await Promise.all(Array.from({ length: SESSIONS }, () => openSession(browser)));

  const expected: Record<string, string> = {};
  await Promise.all(pages.map(async (page, i) => {
    for (const col of COLUMNS) {
      const cell = `${col}${i + 1}`;
      expected[cell] = `s${i + 1}-${col}`;
      await typeInCell(page, cell, expected[cell]);
    }
  }));
  expect(Object.keys(expected)).toHaveLength(SESSIONS * COLUMNS.length);

  // Tras la última acción de cada sesión, el servidor debe tener las 20 celdas.
  const saved: Record<string, string> = {};
  for (const cell of Object.keys(expected)) {
    const res = await request.get(`http://localhost:3001/api/sheets/${SHEET}/cells/${cell}`, { headers: { 'x-user': 'agente@hojas.test' } });
    saved[cell] = (await res.json()).value;
  }
  expect(saved).toEqual(expected);
});
EOF
