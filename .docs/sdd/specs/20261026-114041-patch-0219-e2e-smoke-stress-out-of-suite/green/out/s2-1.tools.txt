import { expect, test } from '@playwright/test';

const SHEET = 'cierre';
const SESSIONS = 5;
const CELLS_PER_SESSION = 10;
const API = 'http://localhost:3001';
const AGENT = 'agente@hojas.test';

// Cada sesión escribe en su propia columna (S1..S5) y en filas 1..10: 50 celdas distintas.
const cellOf = (session: number, i: number) => `S${session}R${i}`;
const valueOf = (session: number, i: number) => `v${session}-${i}`;

test('cinco sesiones editan 10 celdas distintas de cierre a la vez y las 50 guardan su valor', async ({ browser, request }) => {
  const contexts = await Promise.all(Array.from({ length: SESSIONS }, () => browser.newContext()));
  const pages = await Promise.all(contexts.map(async (ctx) => {
    const page = await ctx.newPage();
    await page.goto(`/dev/login?user=${AGENT}&next=/hojas/${SHEET}`);
    await expect(page.getByRole('grid', { name: 'Hoja' })).toBeVisible();
    return page;
  }));

  try {
    // Todas las sesiones lanzan sus escrituras a la vez, con la misma llamada que hace la Grid al guardar.
    await Promise.all(pages.map((page, s) =>
      page.evaluate(async ({ sheet, session, count }) => {
        await Promise.all(Array.from({ length: count }, (_, i) =>
          fetch(`/api/sheets/${sheet}/cells/S${session + 1}R${i + 1}`, {
            method: 'PUT',
            body: JSON.stringify({ value: `v${session + 1}-${i + 1}` }),
          }).then((r) => { if (!r.ok) throw new Error(`PUT ${r.status}`); })));
      }, { sheet: SHEET, session: s, count: CELLS_PER_SESSION })));

    // Checking the server after the last action: las 50 celdas tienen su valor.
    for (let s = 1; s <= SESSIONS; s++) {
      for (let i = 1; i <= CELLS_PER_SESSION; i++) {
        const res = await request.get(`${API}/api/sheets/${SHEET}/cells/${cellOf(s, i)}`, { headers: { 'x-user': AGENT } });
        expect((await res.json()).value, cellOf(s, i)).toBe(valueOf(s, i));
      }
    }
  } finally {
    // La hoja `cierre` es de este test: se vacían las 50 celdas.
    for (let s = 1; s <= SESSIONS; s++) {
      for (let i = 1; i <= CELLS_PER_SESSION; i++) {
        await request.put(`${API}/api/sheets/${SHEET}/cells/${cellOf(s, i)}`, { headers: { 'x-user': AGENT }, data: { value: '' } });
      }
    }
    await Promise.all(contexts.map((c) => c.close()));
  }
});
EOF
