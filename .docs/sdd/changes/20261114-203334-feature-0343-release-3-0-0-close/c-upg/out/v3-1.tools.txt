function core() {
  const lines = ops.split('\n');
  const titles = new Set();
  for (const l of lines) {
    const m = l.match(/^ {2}- (?!\*\*)(.+)$/);
    if (m && m[1].includes(' · ')) titles.add(m[1].split(' · ').pop().trim());

}

}
