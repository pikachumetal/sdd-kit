function titles(config) {
  const out = execFileSync('npx', ['playwright', 'test', '--list', '--reporter=json', '-c', config], {
    encoding: 'utf8',
    shell: true,
    env: { ...process.env, PW_GREP: '', PW_GREP_INVERT: '' },
