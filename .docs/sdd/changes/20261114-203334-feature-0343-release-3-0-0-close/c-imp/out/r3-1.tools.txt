const rates = { Norte: 20, Sur: 15 };

export function priceFor(room, day, hours) {
  if (!(room in rates)) return `sala desconocida: ${room}`;
  const surcharge = day === 'sáb' || day === 'dom' ? weekendSurcharge : 1;
  return `${Math.round(rates[room] * Number(hours) * surcharge)} €`;
}
