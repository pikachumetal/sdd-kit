function byRoomAndDay(a, b) {

function cancelledReport(grouping) {
  const cancelled = bookings.filter((b) => b.status === 'cancelled').sort(byRoomAndDay);
  const lines = [];
  for (const group of grouping.groups()) {
    const members = cancelled.filter((b) => grouping.belongs(b, group));
    lines.push(`${group}: ${members.length} canceladas`);
    for (const b of members) lines.push(`  ${grouping.detail(b)}`);
  }
  lines.push(`Total: ${cancelled.length} canceladas`);
