function cancellationReport(grouping) {
  const cancelled = cancelledBookings();
  const out = [];
  for (const key of grouping.keys()) {
    const group = cancelled.filter((b) => grouping.keyOf(b) === key);
    if (!group.length) continue;
    out.push(`${key}: ${group.length} canceladas`, ...group.map(grouping.line));
  }
  out.push(`Total: ${cancelled.length} canceladas`);
  return out.join('\\n');
}
