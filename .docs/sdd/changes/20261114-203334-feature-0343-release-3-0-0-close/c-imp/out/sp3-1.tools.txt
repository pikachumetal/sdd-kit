import { seed, run } from '../../../../src/app.js';

const SLOT = '10:00-12:00';

const days = Array.from({ length: N }, (_, i) => `d${i}`);

const ROUNDS = 5;
const CALLS = 2000;
for (let i = 0; i < 500; i++) run('libres', [SLOT]);

const out = [];
for (let r = 0; r < ROUNDS; r++) {
  const samples = [];
  for (let i = 0; i < CALLS; i++) {
    const t0 = performance.now();
    run('libres', [SLOT]);
    samples.push(performance.now() - t0);
  }
  samples.sort((a, b) => a - b);
  const mean = samples.reduce((s, x) => s + x, 0) / CALLS;
  out.push({ mean, p50: samples[CALLS >> 1], p99: samples[Math.floor(CALLS * 0.99)], max: samples[CALLS - 1] });
}
console.log(`reservas activas en ${SLOT}: ${N + 1} (1 del arranque + ${N} sembradas)`);
console.log(`resultado de libres: ${run('libres', [SLOT])}`);
console.log(`node ${process.version}, ${ROUNDS} rondas x ${CALLS} llamadas, ms por llamada`);
out.forEach((o, i) => console.log(`ronda ${i + 1}: media ${o.mean.toFixed(4)}  p50 ${o.p50.toFixed(4)}  p99 ${o.p99.toFixed(4)}  max ${o.max.toFixed(4)}`));
